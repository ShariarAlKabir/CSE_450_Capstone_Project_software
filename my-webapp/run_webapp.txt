# Run the web app step by step
# Open one terminal for each section below.

# --------------------------------------------------
# 1) Go to project root
# --------------------------------------------------
cd my-webapp

# --------------------------------------------------
# 2) Start PostgreSQL database (local Homebrew install)
# --------------------------------------------------
# Make sure PostgreSQL is running:
brew services start postgresql@16

# Create the database (first time only):
psql -h /tmp -U dipsaha -d postgres -c "CREATE DATABASE fabric_fault_detection;"

# Run the table creation and data insertion scripts (first time only):
psql -h /tmp -U dipsaha -d fabric_fault_detection -f backend/create_table.sql
psql -h /tmp -U dipsaha -d fabric_fault_detection -f backend/insert_table.sql

# If you need to reset the database later:
psql -h /tmp -U dipsaha -d fabric_fault_detection -f backend/init_db.sql

# --------------------------------------------------
# 3) Start the backend (FastAPI)
# --------------------------------------------------
# Open a new terminal and run:
cd my-webapp/backend
source .venv/bin/activate
pip install -r requirements.txt
uvicorn app.main:app --reload --host 0.0.0.0 --port 8000

# Backend URL:
# http://localhost:8000
# Test it with:
# curl http://localhost:8000/

# --------------------------------------------------
# 4) Start the frontend (React + Vite)
# --------------------------------------------------
# Open another new terminal and run:
cd my-webapp/frontend
npm install
npm run dev -- --host 0.0.0.0

# Frontend URL:
# http://localhost:5173

# --------------------------------------------------
# 5) Useful notes
# --------------------------------------------------
# - The backend connects to PostgreSQL via Unix socket (/tmp) with user dipsaha
# - Database name: fabric_fault_detection
# - Frontend usually runs on port 5173
# - If port 5173 is busy, run:
# npm run dev -- --host 0.0.0.0 --port 5174
